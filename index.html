Quick examples

List available files

import os

print(os.listdir("."))

Read a CSV file

import pandas as pd

df = pd.read_csv("student.csv")

print(df.head())
print(df.describe())

Plot data

import matplotlib.pyplot as plt

plt.scatter(df["energy"], df["dipole"])
plt.xlabel("Energy")
plt.ylabel("Dipole moment")
plt.tight_layout()
plt.savefig("plot.pdf", bbox_inches="tight")
plt.show()
